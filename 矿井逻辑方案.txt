-X		X
	Z

矿井方块坐标：[x,y,z]
矿区范围：(L,W)

西北角坐标：posNW=[x-(L-1)/2,y,z-(W-1)/2]
东南角坐标：posSE=[x+(L-1)/2,y,z+(W-1)/2]


周期C
周期中第n条边（北0->西1->南2->东3）

南北高低落差 hofNS=L-5
东西高低落差 hofWE=W-5

周期内高度偏差 h
实际坐标 H=y-(hofNS+hofWE)*2*C-h
if(n>=1) H-=hofNS
if(n>=2) H-=hofWE
if(n==3) H-=hofNS

南北半边长度halfL=(L-1)/2
东西半边长度halfW=(W-1)/2

if(n%2==0)
{
	idx=0的时候周期内高度偏移h=idx
	idx=1~halfL-1的时候h=idx-1
	idx=halfL（中心位置）的时候h=idx-2
	idx=halfL+1~L-3时h=idx-3
	if(idx>L-3) n+=1
}
else
{
	idx=0的时候周期内高度偏移h=idx
	idx=1~halfW-1的时候h=idx-1
	idx=halfW（中心位置）的时候h=idx-2
	idx=halfW+1~W-3时h=idx-3
	if(idx>W-3) n+=1
	if(n>3)C+=1,n=0
}

北向 n=0
idx=0时的保留区为[posSE.x, H, posNW.z]~[posSE.x, H, posNW.z+1]
idx=1~L-3时的保留区为[posSE.x-idx, H, posNW.z]~[posSE.x-idx+1, H, posNW.z+1]

西向 n=1
idx=0时的保留区为[posNW.x, H, posNW.z]~[posNW.x+1, H, posNW.z]
idx=1~W-3时的保留区为[posNW.x, H, posNW.z+idx]~[posNW.x+1, H, posNW.z+idx-1]

南向 n=2
idx=0时的保留区为[posNW.x, H, posSE.z]~[posNW.x, H, posSE.z-1]
idx=1~L-3时的保留区为[posNW.x+idx, H, posSE.z]~[posNW.x+idx-1, H, posSE.z-1]

东向 n=3
idx=0时的保留区为[posSE.x, H, posSE.z]~[posSE.x-1, H, posSE.z]
idx=1~W-3时的保留区为[posSE.x, H, posSE.z-idx]~[posSE.x-1, H, posSE.z-idx+1]
